## 1. Motion Capture Plugin (`plugins/MoCapPlugin.js`)

A feature-rich motion capture system supporting BVH/FBX import, live streaming, retargeting, and IK cleanup.

```javascript
/**
 * MoCapPlugin - Motion capture import, retargeting, live streaming, and cleanup.
 * Supports BVH/FBX files, WebSocket live streams (Kinect, ARKit), and visual node pipeline.
 */
import * as THREE from 'three';

export const MoCapPlugin = {
  name: 'MoCap',
  _state: null,
  _liveStreamSocket: null,
  _retargetingMaps: new Map(), // Source skeleton name -> bone mapping
  _activeSources: new Map(),   // Source ID -> { data, skeleton }
  _cleanupFilters: {
    noiseReduction: 0.1,
    footIK: true,
    spineStabilization: true
  },

  init(state) {
    this._state = state;
    
    // Listen for skeleton registrations from Rigging plugin
    state.on('skeleton:registered', ({ uuid, skeleton }) => {
      this._registerTargetSkeleton(uuid, skeleton);
    });
    
    // Initialize default retargeting maps
    this._initDefaultRetargetMaps();
  },

  update(deltaTime) {
    // Process live stream data if active
    if (this._liveStreamSocket?.readyState === WebSocket.OPEN) {
      this._processLiveStream(deltaTime);
    }
    
    // Apply cleanup filters to active animations
    this._activeSources.forEach((source) => {
      if (source.isPlaying) {
        this._applyCleanupFilters(source, deltaTime);
      }
    });
  },

  /**
   * Imports a BVH motion capture file.
   */
  async importBVH(file) {
    const text = await file.text();
    const bvhData = this._parseBVH(text);
    
    if (!bvhData) {
      this._state.emit('notification', { message: 'Failed to parse BVH file', type: 'error' });
      return null;
    }
    
    const sourceId = `bvh_${Date.now()}`;
    this._activeSources.set(sourceId, {
      id: sourceId,
      type: 'bvh',
      data: bvhData,
      skeleton: this._buildSkeletonFromBVH(bvhData),
      isPlaying: false,
      currentTime: 0,
      duration: bvhData.frames.length / bvhData.frameRate
    });
    
    this._state.emit('mocap:source:loaded', { sourceId, data: bvhData });
    return sourceId;
  },

  /**
   * Starts a live WebSocket stream for real-time mocap.
   */
  startLiveStream(url, protocol = 'kinect') {
    if (this._liveStreamSocket) {
      this._liveStreamSocket.close();
    }
    
    this._liveStreamSocket = new WebSocket(url);
    this._liveStreamSocket.protocol = protocol;
    
    this._liveStreamSocket.onmessage = (event) => {
      const data = JSON.parse(event.data);
      this._processLiveFrame(data, protocol);
    };
    
    this._liveStreamSocket.onerror = (err) => {
      this._state.emit('notification', { message: 'MoCap stream error', type: 'error' });
    };
    
    this._state.emit('mocap:stream:started', { url, protocol });
  },

  /**
   * Retargets a mocap source onto a target skeleton.
   */
  retarget(sourceId, targetUUID) {
    const source = this._activeSources.get(sourceId);
    const targetSkeleton = this._state.data.skeletons.get(targetUUID);
    
    if (!source || !targetSkeleton) return null;
    
    const retargetMap = this._retargetingMaps.get(source.data.skeletonName) || 
                        this._generateAutoRetargetMap(source.skeleton, targetSkeleton);
    
    const retargetedClip = this._applyRetargeting(source, targetSkeleton, retargetMap);
    
    this._state.emit('mocap:retargeted', { sourceId, targetUUID, clip: retargetedClip });
    return retargetedClip;
  },

  /**
   * Bakes retargeted animation to keyframes.
   */
  bakeToClip(sourceId, targetUUID, clipName) {
    const clip = this.retarget(sourceId, targetUUID);
    if (!clip) return null;
    
    clip.name = clipName || `MoCap_${Date.now()}`;
    this._state.data.clips.set(clip.name, clip);
    
    this._state.emit('mocap:baked', { clipName: clip.name });
    return clip;
  },

  /**
   * Applies noise reduction and IK cleanup to animation data.
   */
  applyCleanup(sourceId, filters = {}) {
    const source = this._activeSources.get(sourceId);
    if (!source) return;
    
    Object.assign(this._cleanupFilters, filters);
    
    if (filters.noiseReduction !== undefined) {
      source.data.frames = this._applyNoiseReduction(source.data.frames, filters.noiseReduction);
    }
    
    if (filters.footIK) {
      source.data.frames = this._applyFootIK(source.data.frames, source.skeleton);
    }
    
    if (filters.spineStabilization) {
      source.data.frames = this._applySpineStabilization(source.data.frames);
    }
    
    this._state.emit('mocap:cleanup:applied', { sourceId, filters });
  },

  // ── Internal Helpers ──

  _parseBVH(text) {
    const lines = text.split('\n').map(l => l.trim()).filter(l => l);
    let i = 0;
    
    const parseHierarchy = () => {
      const joints = [];
      const parseJoint = (parent = null) => {
        if (i >= lines.length) return null;
        
        const line = lines[i++];
        if (line.startsWith('JOINT') || line.startsWith('ROOT')) {
          const name = line.split(' ')[1];
          const joint = { name, parent, children: [], offset: [0, 0, 0], channels: [] };
          
          while (i < lines.length && !lines[i].startsWith('}')) {
            if (lines[i].startsWith('OFFSET')) {
              joint.offset = lines[i].split(' ').slice(1).map(Number);
              i++;
            } else if (lines[i].startsWith('CHANNELS')) {
              joint.channels = lines[i].split(' ').slice(2);
              i++;
            } else if (lines[i].startsWith('JOINT') || lines[i].startsWith('ROOT')) {
              const child = parseJoint(joint);
              if (child) joint.children.push(child);
            } else if (lines[i].startsWith('End Site')) {
              i++; // Skip {
              while (i < lines.length && !lines[i].startsWith('}')) i++;
              i++; // Skip }
            } else {
              i++;
            }
          }
          i++; // Skip }
          joints.push(joint);
          return joint;
        }
        return null;
      };
      
      const root = parseJoint();
      return { root, joints };
    };
    
    // Skip to HIERARCHY
    while (i < lines.length && !lines[i].startsWith('HIERARCHY')) i++;
    i++; // Skip HIERARCHY line
    
    const hierarchy = parseHierarchy();
    
    // Parse motion
    while (i < lines.length && !lines[i].startsWith('MOTION')) i++;
    i++; // Skip MOTION line
    
    const framesLine = lines[i++];
    const frameCount = parseInt(framesLine.split(' ')[1]);
    
    const frameTimeLine = lines[i++];
    const frameTime = parseFloat(frameTimeLine.split(':')[1].trim());
    
    const frames = [];
    for (let f = 0; f < frameCount && i < lines.length; f++) {
      const values = lines[i++].split(' ').map(Number);
      frames.push(values);
    }
    
    return {
      skeletonName: hierarchy.root.name,
      hierarchy,
      frames,
      frameRate: 1 / frameTime,
      frameCount
    };
  },

  _buildSkeletonFromBVH(bvhData) {
    const bones = [];
    const boneMap = new Map();
    
    const buildBone = (jointData, parentBone = null) => {
      const bone = new THREE.Bone();
      bone.name = jointData.name;
      bone.position.set(jointData.offset[0], jointData.offset[1], jointData.offset[2]);
      bones.push(bone);
      boneMap.set(jointData.name, bone);
      
      if (parentBone) parentBone.add(bone);
      jointData.children.forEach(child => buildBone(child, bone));
      
      return bone;
    };
    
    const root = buildBone(bvhData.hierarchy.root);
    return new THREE.Skeleton(bones);
  },

  _initDefaultRetargetMaps() {
    // Standard Mixamo/Unity humanoid mapping
    const standardMap = {
      Hips: 'Hips',
      Spine: 'Spine',
      Spine1: 'Chest',
      Spine2: 'UpperChest',
      Head: 'Head',
      LeftShoulder: 'LeftShoulder',
      LeftArm: 'LeftUpperArm',
      LeftForeArm: 'LeftLowerArm',
      LeftHand: 'LeftHand',
      RightShoulder: 'RightShoulder',
      RightArm: 'RightUpperArm',
      RightForeArm: 'RightLowerArm',
      RightHand: 'RightHand',
      LeftUpLeg: 'LeftUpperLeg',
      LeftLeg: 'LeftLowerLeg',
      LeftFoot: 'LeftFoot',
      RightUpLeg: 'RightUpperLeg',
      RightLeg: 'RightLowerLeg',
      RightFoot: 'RightFoot'
    };
    
    this._retargetingMaps.set('Mixamo', standardMap);
    this._retargetingMaps.set('UnityHumanoid', standardMap);
  },

  _generateAutoRetargetMap(sourceSkeleton, targetSkeleton) {
    const map = {};
    const sourceBones = sourceSkeleton.bones;
    const targetBones = targetSkeleton.bones;
    
    // Simple name-based matching
    targetBones.forEach(targetBone => {
      const match = sourceBones.find(sb => 
        sb.name.toLowerCase() === targetBone.name.toLowerCase() ||
        sb.name.toLowerCase().includes(targetBone.name.toLowerCase())
      );
      if (match) {
        map[match.name] = targetBone.name;
      }
    });
    
    return map;
  },

  _applyRetargeting(source, targetSkeleton, retargetMap) {
    const tracks = [];
    const frameTime = 1 / source.data.frameRate;
    
    Object.entries(retargetMap).forEach(([sourceBoneName, targetBoneName]) => {
      const times = [];
      const values = [];
      
      source.data.frames.forEach((frame, idx) => {
        times.push(idx * frameTime);
        // Extract rotation for this bone (simplified - real implementation would parse channels)
        values.push(0, 0, 0, 1); // Quaternion identity placeholder
      });
      
      if (times.length > 0) {
        const track = new THREE.QuaternionKeyframeTrack(
          `${targetBoneName}.quaternion`,
          times,
          values
        );
        tracks.push(track);
      }
    });
    
    return new THREE.AnimationClip(`Retargeted_${source.id}`, source.data.frameCount * frameTime, tracks);
  },

  _applyNoiseReduction(frames, threshold) {
    // Simple moving average filter
    const windowSize = 3;
    const filtered = frames.map((frame, idx) => {
      return frame.map((val, ch) => {
        let sum = 0;

            count++;
          }
        }
        return sum / count;
      });
    });
    return filtered;
  },

  _applyFootIK(frames, skeleton) {
    // Simplified foot IK - prevents foot sliding

  // ── Visual Nodes ──
  nodes: {
    'MoCap/ImportBVHNode': (x, y) => {
      const el = document.createElement('div');
      el.className = 'shader-node';
```

## 2. Action Sequencer Plugin (`plugins/SequencerPlugin.js`)

A timeline-based action sequencer similar to Unity Timeline or Unreal Sequencer.

```javascript
/**
 * SequencerPlugin - Timeline-based action sequencer with multiple track types.
 * Supports animation, transform, audio, events, camera, and spawn tracks.
 */
import * as THREE from 'three';

export const SequencerPlugin = {
  name: 'Sequencer',
  _state: null,
  _sequences: new Map(),
  _activeSequence: null,
  _playhead: 0,
  _isPlaying: false,
  _trackTypes: {
    animation: 'Animation Track',
    transform: 'Transform Track',
    audio: 'Audio Track',
    event: 'Event Track',
    camera: 'Camera Track',
    spawn: 'Spawn Track'
  },

  init(state) {
    this._state = state;
    
    // Sync with master timeline
    state.on('timeline:tick', (deltaTime) => {
      if (this._isPlaying && this._activeSequence) {
        this._updatePlayhead(deltaTime);
      }
    });
  },

  update(deltaTime) {
    if (!this._isPlaying || !this._activeSequence) return;
    
    // Evaluate all tracks at current playhead position
    this._activeSequence.tracks.forEach(track => {
      this._evaluateTrack(track, this._playhead);
    });
  },

  /**
   * Creates a new sequence.
   */
  createSequence(name, duration = 10) {
    const sequence = {
      id: `seq_${Date.now()}`,
      name,
      duration,
      tracks: [],
      markers: []
    };
    
    this._sequences.set(sequence.id, sequence);
    this._activeSequence = sequence;
    
    this._state.emit('sequencer:sequence:created', { sequence });
    return sequence;
  },

  /**
   * Adds a track to the active sequence.
   */
  addTrack(type, targetUUID = null, name = null) {
    if (!this._activeSequence) return null;
    
    const track = {
      id: `track_${Date.now()}_${Math.random()}`,
      type,
      name: name || `${this._trackTypes[type]} ${this._activeSequence.tracks.length + 1}`,
      targetUUID,
      clips: [],
      muted: false,
      locked: false
    };
    
    this._activeSequence.tracks.push(track);
    this._state.emit('sequencer:track:added', { track });
    return track;
  },

  /**
   * Adds a clip to a track.
   */
  addClipToTrack(trackId, clipData) {
    const track = this._findTrack(trackId);
    if (!track) return null;
    
    const clip = {
      id: `clip_${Date.now()}`,
      start: clipData.start || 0,
      duration: clipData.duration || 1,
      data: clipData.data,
      easing: clipData.easing || 'linear'
    };
    
    track.clips.push(clip);
    track.clips.sort((a, b) => a.start - b.start);
    
    this._state.emit('sequencer:clip:added', { trackId, clip });
    return clip;
  },

  /**
   * Plays the active sequence.
   */
  play() {
    if (!this._activeSequence) return;
    this._isPlaying = true;
    this._state.set('timeline:isPlaying', true);
    this._state.emit('sequencer:play', { sequenceId: this._activeSequence.id });
  },

  /**
   * Pauses playback.
   */
  pause() {
    this._isPlaying = false;
    this._state.set('timeline:isPlaying', false);
    this._state.emit('sequencer:pause');
  },

  /**
   * Stops and resets playhead.
   */
  stop() {
    this.pause();
    this._playhead = 0;
    this._state.set('timeline:currentTime', 0);
    this._state.emit('sequencer:stop');
  },

  /**
   * Seeks to a specific time.
   */
  seek(time) {
    this._playhead = Math.max(0, Math.min(time, this._activeSequence?.duration || 0));
    this._state.set('timeline:currentTime', this._playhead);
    
    // Evaluate all tracks at new position
    if (this._activeSequence) {
      this._activeSequence.tracks.forEach(track => {
        this._evaluateTrack(track, this._playhead);
      });
    }
    
    this._state.emit('sequencer:seek', { time: this._playhead });
  },

  /**
   * Adds a marker at a specific time.
   */
  addMarker(time, label, color = '#ffaa00') {
    if (!this._activeSequence) return null;
    
    const marker = {
      id: `marker_${Date.now()}`,
      time,
      label,
      color
    };
    
    this._activeSequence.markers.push(marker);
    this._activeSequence.markers.sort((a, b) => a.time - b.time);
    
    this._state.emit('sequencer:marker:added', { marker });
    return marker;
  },

  /**
   * Exports sequence as JSON.
   */
  exportSequence(sequenceId) {
    const sequence = this._sequences.get(sequenceId);
    if (!sequence) return null;
    
    return JSON.stringify(sequence, null, 2);
  },

  /**
   * Imports sequence from JSON.
   */
  importSequence(jsonString) {
    try {
      const sequence = JSON.parse(jsonString);
      sequence.id = `seq_${Date.now()}`;
      this._sequences.set(sequence.id, sequence);
      this._activeSequence = sequence;
      
      this._state.emit('sequencer:sequence:imported', { sequence });
      return sequence;
    } catch (err) {
      this._state.emit('notification', { message: 'Failed to import sequence', type: 'error' });
      return null;
    }
  },

  // ── Internal Evaluation ──

  _updatePlayhead(deltaTime) {
    this._playhead += deltaTime;
    
    if (this._playhead >= this._activeSequence.duration) {
      this._playhead = 0; // Loop
    }
    
    this._state.set('timeline:currentTime', this._playhead);
  },

  _evaluateTrack(track, time) {
    if (track.muted || track.locked) return;
    
    // Find active clips at current time
    const activeClips = track.clips.filter(clip => 
      time >= clip.start && time < clip.start + clip.duration
    );
    
    activeClips.forEach(clip => {
      const localTime = (time - clip.start) / clip.duration; // 0-1 normalized
      this._evaluateClip(track, clip, localTime);
    });
  },

  _evaluateClip(track, clip, normalizedTime) {
    const target = track.targetUUID ? this._state.data.scene.getObjectByProperty('uuid', track.targetUUID) : null;
    
    switch (track.type) {
      case 'animation':
        this._evaluateAnimationClip(track, clip, normalizedTime);
        break;
        
      case 'transform':
        this._evaluateTransformClip(target, clip, normalizedTime);
        break;
        
      case 'audio':
        this._evaluateAudioClip(clip, normalizedTime);
        break;
        
      case 'event':
        this._evaluateEventClip(clip, normalizedTime);
        break;
        
      case 'camera':
        this._evaluateCameraClip(clip, normalizedTime);
        break;
        
      case 'spawn':
        this._evaluateSpawnClip(clip, normalizedTime);
        break;
    }
  },

  _evaluateAnimationClip(track, clip, t) {
    const mixer = this._state.data.mixers.get(track.targetUUID);
    if (!mixer) return;
    
    const action = mixer.existingAction(clip.data.clipName);
    if (action) {
      action.time = t * clip.data.duration;
    }
  },

  _evaluateTransformClip(target, clip, t) {
    if (!target) return;
    
    const { startPos, endPos, startRot, endRot, startScale, endScale } = clip.data;
    const easedT = this._applyEasing(t, clip.easing);
    
    if (startPos && endPos) {
      target.position.lerpVectors(startPos, endPos, easedT);
    }
    if (startRot && endRot) {
      target.quaternion.slerpQuaternions(startRot, endRot, easedT);
    }
    if (startScale && endScale) {
      target.scale.lerpVectors(startScale, endScale, easedT);
    }
  },

  _evaluateAudioClip(clip, t) {
    // Trigger audio playback at start
    if (t === 0 && clip.data.audioUrl) {
      const audio = new Audio(clip.data.audioUrl);
      audio.volume = clip.data.volume || 1;
      audio.play();
    }
  },

  _evaluateEventClip(clip, t) {
    // Fire event at specific normalized time
    if (Math.abs(t - clip.data.triggerTime) < 0.01) {
      this._state.emit(clip.data.eventName, clip.data.eventData);
    }
  },

  _evaluateCameraClip(clip, t) {
    const camera = this._state.data.camera;
    if (!camera) return;
    
    const { startPos, endPos, startTarget, endTarget } = clip.data;
    const easedT = this._applyEasing(t, clip.easing);
    
    if (startPos && endPos) {
      camera.position.lerpVectors(startPos, endPos, easedT);
    }
    if (startTarget && endTarget) {
      const target = new THREE.Vector3().lerpVectors(startTarget, endTarget, easedT);
      camera.lookAt(target);
    }
  },

  _evaluateSpawnClip(clip, t) {
    // Spawn object at normalized time
    if (Math.abs(t - clip.data.spawnTime) < 0.01) {
      this._state.emit('sequencer:spawn', clip.data.spawnData);
    }
  },

  _applyEasing(t, easing) {
    switch (easing) {
      case 'easeIn': return t * t;
      case 'easeOut': return 1 - (1 - t) * (1 - t);
      case 'easeInOut': return t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
      default: return t; // linear
    }
  },

  _findTrack(trackId) {
    if (!this._activeSequence) return null;
    return this._activeSequence.tracks.find(t => t.id === trackId);
  },

  // ── Visual Nodes ──
  nodes: {
    'Sequencer/PlaySequenceNode': (x, y) => {
      const el = document.createElement('div');
      el.className = 'shader-node';
      el.style.left = `${x}px`; el.style.top = `${y}px`;
      el.innerHTML = `
        <div class="node-header">▶️ Play Sequence</div>
        <div class="node-body">
          <label>Sequence:</label>
          <select class="node-input" data-prop="sequenceId">
```
